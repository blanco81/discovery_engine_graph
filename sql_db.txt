-- Asegurarnos de empezar de cero
DROP TABLE IF EXISTS maintenance_requests, sensor_readings, sensors, access_logs, students, rfid_cards, rooms, buildings CASCADE;

-- 1. Tabla de Edificios (Dominio base)
CREATE TABLE buildings (
    bldg_id SERIAL PRIMARY KEY,
    bldg_name VARCHAR(100),
    zone_id INTEGER -- Usado como trampa (coincidirá con prioridades)
);

-- 2. Tabla de Habitaciones/Aulas
CREATE TABLE rooms (
    room_id SERIAL PRIMARY KEY,
    room_code VARCHAR(20),
    bldg_id INTEGER -- Relación oculta real: rooms.bldg_id -> buildings.bldg_id
);

-- 3. Tabla de Tarjetas RFID
CREATE TABLE rfid_cards (
    card_id SERIAL PRIMARY KEY,
    issue_date DATE
);

-- 4. Tabla de Estudiantes
CREATE TABLE students (
    student_id SERIAL PRIMARY KEY,
    student_name VARCHAR(100),
    active_card_id INTEGER -- Relación oculta real: students.active_card_id -> rfid_cards.card_id
);

-- 5. Tabla de Logs de Acceso (Transaccional)
CREATE TABLE access_logs (
    log_id SERIAL PRIMARY KEY,
    card_id INTEGER,      -- Relación oculta real: access_logs.card_id -> rfid_cards.card_id
    room_id INTEGER,      -- Relación oculta real: access_logs.room_id -> rooms.room_id
    access_time TIMESTAMP
);

-- 6. Tabla de Sensores IoT
CREATE TABLE sensors (
    sensor_id SERIAL PRIMARY KEY,
    room_id INTEGER,      -- Relación oculta real: sensors.room_id -> rooms.room_id
    sensor_type VARCHAR(50)
);

-- 7. Tabla de Lecturas de Sensores (Transaccional masiva)
CREATE TABLE sensor_readings (
    reading_id SERIAL PRIMARY KEY,
    sensor_id INTEGER,    -- Relación oculta real: sensor_readings.sensor_id -> sensors.sensor_id
    reading_value DECIMAL,
    reading_time TIMESTAMP
);

-- 8. Tabla de Mantenimiento (Trampa semántica)
CREATE TABLE maintenance_requests (
    req_id SERIAL PRIMARY KEY,
    room_id INTEGER,      -- Relación oculta real: maintenance_requests.room_id -> rooms.room_id
    priority INTEGER,     -- Trampa: Valores 1-5 que coincidirirá con bldg_id
    status VARCHAR(20)
);
